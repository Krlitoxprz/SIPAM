\chapter{Introduction and Justification}
% ════════════════════════════════════════════════════════════

\section{Institutional Background}

The Universidad Surcolombiana (USCO), located in Neiva, Colombia, manages academic monitoring
programs (\textit{monitorías}) and field practice excursions (\textit{prácticas extramuros}) under
\textbf{Acuerdo 012/2023}. All current processes rely on paper forms (FO-14, FO-46), manual
spreadsheets, and in-person evaluations — creating significant inefficiencies.

\section{Problem Statement}

\begin{enumerate}[label=\textbf{P\arabic*.},leftmargin=*]
\item \textbf{Manual Selection}: Scoring formula from Acuerdo 012/2023 is computed in spreadsheets, prone to errors and manipulation.
\item \textbf{No GPS Traceability}: Field excursions have no real-time location tracking.
\item \textbf{Manual Document Verification}: No automated OCR-based extraction from identity documents.
\item \textbf{Subjective Letter Evaluation}: Motivation letters assessed subjectively without semantic analysis.
\item \textbf{No Predictive Analytics}: No AI-based tools to predict student selection probability.
\item \textbf{Fragmented Workflows}: Form generation, QR attendance, and excursion management are disconnected.
\end{enumerate}

\section{Objectives}

\subsection{General Objective}
Design, develop, and deploy SIPAM-USCO: an intelligent web platform that automates the management
of monitoring programs and field practices at USCO, integrating ML, CNN, NLP, and IoT GPS.

\subsection{Specific Objectives}
\begin{enumerate}[label=\textbf{SO\arabic*.},leftmargin=*]
\item Implement a transactional backend with RBAC using FastAPI and JWT (6 roles).
\item Build an AI microservice (Flask) with 3+ ML models for selection prediction.
\item Integrate CNN OCR (TensorFlow/Keras) for identity document text extraction.
\item Implement Transformer NLP (DistilBERT) for motivation letter semantic analysis.
\item Design IoT ecosystem (ESP32 + NEO-6M) for real-time GPS tracking of field practices.
\item Deploy on AWS EC2 with Docker Compose and CI/CD via GitHub.
\item Validate with unit, functional and integration tests with $\geq 85\%$ coverage.
\end{enumerate}

The Acuerdo 012/2023 from USCO's Superior Council requires: GPA $\geq 3.5$, at least 40\% of

\begin{equation}
Score = 0.4 \times \bar{x}_{acad} + 0.3 \times \%_{cred} + 0.2 \times grade_{subj} + 0.1 \times grade_{interv}
\end{equation}

\text{Attention}(Q,K,V)=\text{softmax}\!\left(\tfrac{QK^T}{\sqrt{d_k}}\right)V,\quad
\text{MultiHead}=\text{Concat}(\text{head}_1,\ldots,\text{head}_h)\,W^O

\section{AI Requirements}

\begin{reqenv}{RF-IA-01}{ML Ensemble Prediction}
\reqrow{Code}{RF-IA-01}
\reqrow{Type}{AI — Binary Classification}
\reqrow{Priority}{\textcolor{uscoRed}{\textbf{High}}}
\reqrow{Description}{Flask microservice (port 5001) computes selection probability (0--100\%) using pre-trained Random Forest with SMOTE balancing and Optuna hyperparameter tuning.}
\reqrow{Models}{MLP Neural Network, Decision Tree, Random Forest (3+ models required by rubric)}
\reqrow{Data Split}{70\% training / 15\% validation / 15\% test}
\reqrow{Serialization}{Joblib for loading without retraining}
\reqrow{Acceptance Criteria}{Inference $\leq 2$ s. SHAP feature importance. F1-Score $\geq 0.80$.}
\end{reqenv}

\begin{reqenv}{RF-OCR-01}{Computer Vision — CNN OCR}
\reqrow{Code}{RF-OCR-01}
\reqrow{Type}{AI — Computer Vision}
\reqrow{Priority}{Medium}
\reqrow{Description}{Custom CNN (TensorFlow/Keras) with Conv2D×4 architecture automatically extracts alphanumeric text from identity document images.}
\reqrow{Architecture}{Conv2D(32) → BN → Pool → Conv2D(64) → BN → Pool → Conv2D(128) → BN → Pool → Conv2D(256) → GAP → Dense(512) → Dense(37)}
\reqrow{Pre-processing}{Grayscale + resize 128×128 + normalization [0,1]}
\reqrow{Acceptance Criteria}{37 classes (A--Z, 0--9, blank). Confidence score returned; if $< 75\%$, escalate to manual review. Accuracy $\geq 90\%$.}
\end{reqenv}

\begin{reqenv}{RF-NLP-01}{Transformer NLP — Letter Analysis}
\reqrow{Code}{RF-NLP-01}
\reqrow{Type}{AI — NLP}
\reqrow{Priority}{\textcolor{uscoRed}{\textbf{High}}}
\reqrow{Description}{DistilBERT-base-uncased evaluates motivation letters extracting: quality score (0--2), sentiment (Positive/Neutral/Negative), coherence (0--1), key phrases, attention weights.}
\reqrow{Acceptance Criteria}{Max token length: 512. Batch analysis up to 50 texts. Response time $\leq 5$ s.}
\end{reqenv}

\begin{reqenv}{RF-IOT-01}{GPS Telemetry — ESP32 + NEO-6M}
\reqrow{Code}{RF-IOT-01 to RF-IOT-07}
\reqrow{Type}{IoT — Hardware + Software}
\reqrow{Priority}{\textcolor{uscoRed}{\textbf{High}}}
\reqrow{Hardware}{ESP32 DevKit V1 (dual-core 240 MHz, WiFi 802.11 b/g/n) + u-blox NEO-6M (2.5 m CEP, 1 Hz, NMEA 0183)}
\reqrow{Protocol}{HTTP/1.1 REST POST every 60 s; X-Api-Key authentication}
\reqrow{GPS Fields}{latitude, longitude, altitude, speed\_kmh, course, satellites, hdop, datetime\_gps}
\reqrow{Telemetry Fields}{battery\_voltage, wifi\_rssi, uptime\_seconds}
\reqrow{Acceptance Criteria}{WiFi auto-reconnect with exponential backoff. GeoJSON route export. Public \texttt{/iot/health} endpoint.}
\end{reqenv}
