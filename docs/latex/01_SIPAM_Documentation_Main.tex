\chapter{Introduction}

\section{Background}

The Universidad Surcolombiana (USCO) requires a comprehensive system to manage academic monitoring programs and field practice excursions (extramuros) for its engineering faculty. Currently, these processes are managed through paper-based forms and manual tracking, leading to inefficiencies and lack of traceability.

\section{Problem Statement}

\subsection{Current Issues}
\begin{itemize}
    \item \textbf{Manual Process}: Monitor selection follows Acuerdo 012/2023 but lacks digital support
    \item \textbf{No Traceability}: No centralized tracking of practice excursions
    \item \textbf{Security Concerns}: Paper-based attendance and transport requests
    \item \textbf{No Predictive Capability}: No data-driven insights for monitor selection
\end{itemize}

\subsection{Proposed Solution}
SIPAM-USCO digitizes the entire workflow from monitor convocation creation to practice excursion approval, including:
\begin{enumerate}
    \item Digital FO-14, FO-15, FO-16 forms
    \item AI-powered monitor selection prediction
    \item QR-based attendance tracking
    \item Real-time GPS tracking for excursions
\end{enumerate}

\section{Objectives}

\subsection{General Objective}
Develop a web-based system to automate the management of academic monitoring programs and field practice excursions at Universidad Surcolombiana.

\subsection{Specific Objectives}
\begin{enumerate}
    \item Implement user role management (Admin, Decano, Jefe de Programa, Profesor, Estudiante)
    \item Develop convocatoria workflow with state machine (borrador $\rightarrow$ abierta $\rightarrow$ cerrada $\rightarrow$ en\_evaluacion $\rightarrow$ finalizada)
    \item Create AI microservice for selection probability prediction
    \item Implement QR code generation for FO-46 and attendance
    \item Deploy on AWS with Docker containerization
\end{enumerate}

\chapter{State of the Art and Literature Review}

\section{Legal Framework}

\subsection{Acuerdo 012/2023}
The Colombian higher education monitoring framework establishes:
\begin{itemize}
    \item 11 types of monitoring (Art. 3)
    \item Selection criteria based on academic performance (Art. 4)
    \item Minimum 5-day postulation window (Art. 6)
    \item Digital or physical publication of results (Art. 8)
\end{itemize}

\section{Technological Context}

\subsection{Similar Systems}
\begin{itemize}
    \item \textbf{Moodle}: Learning management but no monitoring-specific workflows
    \item \textbf{Academic ERPs}: Generic, lack Colombian regulatory compliance
    \item \textbf{SIPAM}: First Colombian system specifically for Acuerdo 012/2023
\end{itemize}
